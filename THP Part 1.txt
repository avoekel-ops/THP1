1.
SELECT bakers.fullname
FROM bakers
JOIN technicals ON bakers.baker = technicals.baker
WHERE rank = '2';

2.
SELECT bakers.fullname
FROM bakers
JOIN technicals ON bakers.baker = technicals.baker
WHERE rank = '2'
EXCEPT 
SELECT bakers.fullname
FROM bakers
JOIN technicals ON bakers.baker = technicals.baker
WHERE rank = '1';

3.
SELECT bakers.fullname, bakers.hometown
FROM bakers
JOIN signatures on bakers.baker = signatures.baker
WHERE make Like '%Raisin%';

4.
SELECT bakers.fullname, bakers.hometown
FROM bakers
INNER JOIN signatures ON bakers.baker = signatures.baker
LEFT JOIN showstoppers ON signatures.baker = showstoppers.baker
WHERE showstoppers.make LIKE '%Hazelnut%'
OR signatures.make LIKE '%Hazelnut%'
GROUP BY bakers.fullname, bakers.hometown;

5.
SELECT bakers.fullname, results.episodeid
FROM bakers
JOIN results ON bakers.baker = results.baker
WHERE result = 'eliminated'
AND episodeid < 5;

6.
SELECT results.baker, showstoppers.make
FROM results
JOIN showstoppers ON results.baker = showstoppers.baker
AND results.episodeid = showstoppers.episodeid
WHERE result = 'star baker';
